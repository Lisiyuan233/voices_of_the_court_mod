# VOTC NPC-initiated letters: direct controls used by the custom HUD window.
# The GUI root is always the local player character.

votc_incoming_letters_enable_sgui = {
	scope = character
	effect = {
		set_variable = {
			name = votc_incoming_letters_enabled
			value = 1
		}
		if = {
			limit = { NOT = { has_variable = votc_incoming_letters_frequency } }
			set_variable = {
				name = votc_incoming_letters_frequency
				value = 2
			}
		}
		if = {
			limit = { NOT = { has_variable = votc_incoming_letters_event_frequency } }
			set_variable = {
				name = votc_incoming_letters_event_frequency
				value = 2
			}
		}
		if = {
			limit = { NOT = { has_variable = votc_incoming_letters_family_frequency } }
			set_variable = {
				name = votc_incoming_letters_family_frequency
				value = 2
			}
		}
		if = {
			limit = { NOT = { has_variable = votc_incoming_letters_poetry_frequency } }
			set_variable = {
				name = votc_incoming_letters_poetry_frequency
				value = 2
			}
		}
	}
}

votc_incoming_letters_disable_sgui = {
	scope = character
	effect = {
		remove_variable = votc_incoming_letters_enabled
	}
}

votc_battle_reports_enable_sgui = {
	scope = character
	effect = {
		# Absence means enabled so existing saves retain their current behaviour.
		remove_variable ?= votc_battle_reports_disabled
	}
}

votc_battle_reports_disable_sgui = {
	scope = character
	effect = {
		set_variable = {
			name = votc_battle_reports_disabled
			value = 1
		}
	}
}

		}
