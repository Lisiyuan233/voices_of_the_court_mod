# VOTC save-load init: identify the campaign as soon as a save is loaded.
#
# Without this hook the campaign identity is only bootstrapped lazily on the
# first business action (a conversation, an incoming letter, the quarterly
# chronicle pulse). A player who loads an older save and opens a conversation
# right away would close it against a save whose campaign id was still absent,
# and the close path fails closed in that case.
#
# on_game_start_after_lobby already has an effect block in vanilla, and a
# second effect or trigger block for the same on_action is silently ignored
# (vanilla common/on_action/_on_actions.info, MODDING INFO), so the only safe
# way to append work is the official forwarding pattern: register a mod-owned
# relay and do the work there.
#
# The hook's root scope is none, so the player is reached through
# every_player + is_ai = no - the same idiom vanilla uses inside the
# on_game_start block of common/on_action/game_start.txt.
on_game_start_after_lobby = {
	on_actions = {
		votc_game_start_init_relay
	}
}

votc_game_start_init_relay = {
	effect = {
		every_player = {
			limit = { is_ai = no }
			# Idempotent: an intact campaign id and checkpoint are left as they
			# are, a missing one is generated here rather than on the first
			# conversation. The bootstrap kind the effect records says which of
			# those happened (see votc_init_campaign_id_effect).
			votc_init_checkpoint_effect = yes
			# Identity line the app consumes (app: parseCampaignLoadedLine ->
			# campaignLoadObserver): it gives the session its campaign id before the
			# first conversation, which is when the VOTC:IN init line is normally
			# emitted, and bootstrapKind 2/3 makes the app tell the user their save
			# was upgraded instead of leaving them with a silently new campaign id.
			# Field order is part of the contract: player, schema, a, b, c, d,
			# bootstrap kind, checkpoint epoch.
			debug_log = "VOTC:CAMPAIGN/;/loaded/;/[GetPlayer.GetID]/;/[GetGlobalVariable('votc_campaign_schema').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_a').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_b').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_c').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_d').GetValue|0]/;/[GetGlobalVariable('votc_campaign_bootstrap_kind').GetValue|0]/;/[GetPlayer.MakeScope.Var('votc_checkpoint_epoch').GetValue|0]"
		}
	}
}
