# VOTC NPC-initiated letters: shared triggers.
# All scopes are character scopes.

# Player scope: has the feature been enabled via the settings decision.
votc_incoming_letters_enabled_trigger = {
	has_variable = votc_incoming_letters_enabled
}

# Player scope: shared readiness for letter categories that use slots 1-6.
# Siege and lost-holding letters use this directly because their own category
# switch is independent from the "other letters" category.
votc_incoming_letters_slot_ready_trigger = {
	votc_incoming_letters_enabled_trigger = yes
	OR = {
		NOT = { has_variable = votc_incoming_letters_global_cooldown }
		var:votc_incoming_letters_frequency = 4
	}
	OR = {
		NOT = { exists = global_var:votc_incoming_letter_1 }
		NOT = { exists = global_var:votc_incoming_letter_2 }
		NOT = { exists = global_var:votc_incoming_letter_3 }
		NOT = { exists = global_var:votc_incoming_letter_4 }
		NOT = { exists = global_var:votc_incoming_letter_5 }
		NOT = { exists = global_var:votc_incoming_letter_6 }
	}
}

# Player scope: shared readiness plus the "other letters" category switch.
# Used by quarterly and event-driven regular incoming letters.
votc_incoming_letters_ready_trigger = {
	NOT = { var:votc_incoming_letters_frequency = 5 }
	votc_incoming_letters_slot_ready_trigger = yes
}

# Player scope: the dedicated close-family slot and close-family cooldown are
# independent from slots 1-6 and from the shared global cooldown.
votc_incoming_family_letter_ready_trigger = {
	votc_incoming_letters_enabled_trigger = yes
	NOT = { var:votc_incoming_letters_family_frequency = 5 }
	OR = {
		NOT = { has_variable = votc_incoming_letter_family_cooldown }
		var:votc_incoming_letters_family_frequency = 4
	}
	NOT = { exists = global_var:votc_incoming_letter_7 }
}

# Player scope: poetry correspondence uses its own quarterly chance, cooldown
# and dedicated slot. It is independent from both the shared slots and family
# correspondence.
votc_incoming_poetry_letter_ready_trigger = {
	votc_incoming_letters_enabled_trigger = yes
	NOT = { var:votc_incoming_letters_poetry_frequency = 5 }
	OR = {
		NOT = { has_variable = votc_incoming_letter_poetry_cooldown }
		var:votc_incoming_letters_poetry_frequency = 4
	}
	NOT = { exists = global_var:votc_incoming_letter_8 }
}

# Character scope: a valid NPC author for an incoming letter.
# Excludes dead, children, players, prisoners and pool/temp characters, plus recent senders.
votc_incoming_letter_author_valid_trigger = {
	is_alive = yes
	is_adult = yes
	is_ai = yes
	is_imprisoned = no
	is_pool_character = no
	NOT = { has_variable = votc_incoming_letter_sent_recently }
}

# Character scope: a valid close-family author. This uses its own sender
# cooldown, so sending or receiving another category does not suppress family
# correspondence and vice versa.
votc_incoming_letter_family_author_valid_trigger = {
	is_alive = yes
	is_adult = yes
	is_ai = yes
	is_imprisoned = no
	is_pool_character = no
	NOT = { has_variable = votc_incoming_letter_family_sent_recently }
}

# Character scope: a valid author for the dedicated poetry category. Poetry
# correspondence has its own sender cooldown. At maximum frequency, the
# cooldown is ignored immediately even if it was created by an older setting.
votc_incoming_letter_poetry_author_valid_trigger = {
	is_alive = yes
	is_adult = yes
	is_ai = yes
	is_imprisoned = no
	is_pool_character = no
	OR = {
		NOT = { has_variable = votc_incoming_letter_poetry_sent_recently }
		scope:actor = { var:votc_incoming_letters_poetry_frequency = 4 }
	}
}

# Great Project scope: the current player owns, founded, or contributed to this
# native CK3 Great Project. scope:actor must be the player character.
votc_incoming_letter_great_project_related_trigger = {
	OR = {
		great_project_owner = scope:actor
		great_project_founder = scope:actor
		any_contribution = {
			contributor = scope:actor
		}
	}
}

# Great Project scope: milestone-specific readiness for the current player.
# Reported players are stored in variable lists so multiplayer participants
# receive independent reports for the same project.
votc_incoming_letter_great_project_planned_ready_trigger = {
	votc_incoming_letter_great_project_related_trigger = yes
	great_project_construction_phase_is = planned
	NOT = {
		any_in_list = {
			variable = votc_great_project_planned_reported_players
			this = scope:actor
		}
	}
}

votc_incoming_letter_great_project_started_ready_trigger = {
	votc_incoming_letter_great_project_related_trigger = yes
	great_project_construction_phase_is = in_progress
	great_project_build_progress < 0.5
	NOT = {
		any_in_list = {
			variable = votc_great_project_started_reported_players
			this = scope:actor
		}
	}
}
