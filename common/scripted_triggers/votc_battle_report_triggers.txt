votc_battle_report_has_required_scopes_trigger = {
	exists = combat.location
	exists = side_primary_participant
	exists = enemy_side.side_primary_participant
}

## Battle delivery state is held as local variables and later consumed by a
## console run script. Unlanded characters may participate in combat but do
## not provide a persistent variable owner, so they must not become the
## delivery target.
votc_battle_report_variable_owner_trigger = {
	is_ai = no
	is_landed = yes
	NOT = { has_variable = votc_battle_reports_disabled }
}

votc_battle_report_player_involved_trigger = {
	OR = {
		side_primary_participant = { votc_battle_report_variable_owner_trigger = yes }
		enemy_side.side_primary_participant = { votc_battle_report_variable_owner_trigger = yes }
		any_side_participant = { votc_battle_report_variable_owner_trigger = yes }
		enemy_side = {
			any_side_participant = { votc_battle_report_variable_owner_trigger = yes }
		}
		side_commander ?= { votc_battle_report_variable_owner_trigger = yes }
		enemy_side.side_commander ?= { votc_battle_report_variable_owner_trigger = yes }
		any_side_commander = { votc_battle_report_variable_owner_trigger = yes }
		enemy_side = {
			any_side_commander = { votc_battle_report_variable_owner_trigger = yes }
		}
		any_side_knight = { votc_battle_report_variable_owner_trigger = yes }
		enemy_side = {
			any_side_knight = { votc_battle_report_variable_owner_trigger = yes }
		}
	}
}

votc_battle_report_winning_side_player_trigger = {
	OR = {
		side_primary_participant = { votc_battle_report_variable_owner_trigger = yes }
		any_side_participant = { votc_battle_report_variable_owner_trigger = yes }
		side_commander ?= { votc_battle_report_variable_owner_trigger = yes }
		any_side_commander = { votc_battle_report_variable_owner_trigger = yes }
		any_side_knight = { votc_battle_report_variable_owner_trigger = yes }
	}
}

votc_battle_report_important_trigger = {
	OR = {
		exists = scope:wipe
		troops_ratio <= 0.5
		percent_enemies_killed >= 50
		num_enemies_killed >= 3000
		combat = { num_total_troops >= 10000 }
		combat = { warscore_value >= 5 }
		any_in_list = {
			list = slain_combatants
			OR = {
				this = root.side_primary_participant
				this = root.enemy_side.side_primary_participant
				is_heir_of = root.side_primary_participant
				is_heir_of = root.enemy_side.side_primary_participant
			}
		}
		any_in_list = {
			list = prisoners_of_war
			OR = {
				this = root.side_primary_participant
				this = root.enemy_side.side_primary_participant
				is_heir_of = root.side_primary_participant
				is_heir_of = root.enemy_side.side_primary_participant
			}
		}
	}
}
