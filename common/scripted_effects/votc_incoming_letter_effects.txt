# Dedicated close-family selection
# ------------------------------

# Current scope = player. Selects either a spouse or a close blood relative.
# Location is deliberately unrestricted; this is the relaxed family category.
votc_select_family_letter_author_effect = {
	random_list = {
		25 = {
			trigger = {
				any_spouse = {
					votc_incoming_letter_family_author_valid_trigger = yes
				}
			}
			random_spouse = {
				limit = {
					votc_incoming_letter_family_author_valid_trigger = yes
				}
				save_scope_as = recipient
			}
		}
		75 = {
			trigger = {
				any_close_family_member = {
					votc_incoming_letter_family_author_valid_trigger = yes
				}
			}
			random_close_family_member = {
				limit = {
					votc_incoming_letter_family_author_valid_trigger = yes
				}
				save_scope_as = recipient
			}
		}
	}
}

# Current scope = player; scope:actor = player. Candidate priority:
# 1. friends, lovers, soulmates, spouses, close family, or the player's court poet;
# 2. characters with the Poet trait in the player's own sub-realm;
# 3. characters with the Poet trait anywhere under the same top liege.
# Later tiers are only considered when the earlier tier has no valid candidate.
votc_select_poetry_letter_author_effect = {
	remove_variable ?= votc_poetry_sender_source

	if = {
		limit = {
			OR = {
				any_relation = {
					type = friend

				}

					votc_incoming_letter_poetry_author_valid_trigger = yes

					votc_incoming_letter_poetry_author_valid_trigger = yes

				trigger = {
					any_relation = {
						type = friend

# ------------------------------
# Courtier-petition preparation
# ------------------------------

# Current scope = petitioning courtier; scope:actor = player.
# Only eligible petition sub-types participate in the weighted selection. This
# avoids the old behavior where every failed condition became a land request.
votc_prepare_courtier_petition_effect = {
	remove_variable ?= votc_petition_type

	random_list = {
		18 = {
			# Unmarried adult courtiers may ask the liege to arrange a marriage.
			trigger = {
				is_married = no
				age >= 16
			}
			set_variable = { name = votc_petition_type value = flag:marriage_request }
		}
		14 = {
			# Per the intended court convention, women do not petition for land.
			trigger = { is_male = yes }
			set_variable = { name = votc_petition_type value = flag:land_grant }
		}
		12 = {
			# Salaried office requests are restricted to male courtiers.
			trigger = { is_male = yes }
			set_variable = { name = votc_petition_type value = flag:office_request }
		}
		8 = {
			# Do not ask for a council seat when already serving as a councillor.
			trigger = {
				is_male = yes
				is_councillor = no
			}
			set_variable = { name = votc_petition_type value = flag:council_seat }
		}
		8 = {
			# A personal tax/obligation relief request needs a concrete hardship.
			trigger = { stress_level >= 1 }
			set_variable = { name = votc_petition_type value = flag:tax_relief }
		}
		8 = {
			# Honor/title advancement follows the same male-only convention.
			trigger = { is_male = yes }
			set_variable = { name = votc_petition_type value = flag:title_promotion }
		}
		6 = {
			# The player must actually hold a close relative, friend or lover.
			trigger = {
				scope:actor = {
					any_prisoner = {
						OR = {
							is_close_family_of = scope:recipient
							has_relation_friend = scope:recipient
							has_relation_lover = scope:recipient
						}
					}
				}
			}
			set_variable = { name = votc_petition_type value = flag:prisoner_release }
		}
		6 = {
			# The courtier can ask the player to honour a hook they hold.
			trigger = { has_hook = scope:actor }
			set_variable = { name = votc_petition_type value = flag:hook_use }
		}
		5 = {
			# A martial specialist may ask for wartime resources while the liege is at war.
			trigger = {
				highest_skill = martial
				scope:actor = { is_at_war = yes }
			}
			set_variable = { name = votc_petition_type value = flag:war_aid }
		}
		5 = {
			# A courtier with capacity may offer to educate an unguarded minor child.
			trigger = {
				num_of_relation_ward < 2
				scope:actor = {
					any_child = {
						is_adult = no
						num_of_relation_guardian = 0
					}
				}
			}
			set_variable = { name = votc_petition_type value = flag:guardian_request }
		}
		10 = {
			# A general reward remains the universal fallback for any adult courtier.
			set_variable = { name = votc_petition_type value = flag:gift_request }
		}
		5 = {
			# A family marriage proposal requires an unmarried adult child.
			trigger = {
				any_child = {
					is_adult = yes
					is_married = no
				}
			}
			set_variable = { name = votc_petition_type value = flag:alliance_proposal }
		}
	}

	set_variable = {
		name = votc_courtier_years
		value = days_since_joined_court
	}
	change_variable = {
		name = votc_courtier_years
		divide = 365
	}
	round_variable = {
		name = votc_courtier_years
		nearest = 1
	}
	if = {
		limit = { highest_skill = diplomacy }
		set_variable = { name = votc_courtier_skill value = flag:diplomacy }
	}
	else_if = {
		limit = { highest_skill = martial }
		set_variable = { name = votc_courtier_skill value = flag:martial }
	}
	else_if = {
		limit = { highest_skill = stewardship }
		set_variable = { name = votc_courtier_skill value = flag:stewardship }
	}
	else_if = {
		limit = { highest_skill = intrigue }
		set_variable = { name = votc_courtier_skill value = flag:intrigue }
	}
	else = {
		set_variable = { name = votc_courtier_skill value = flag:learning }
	}
}

# Slot allocation
# ------------------------------

# Assumes scope:actor = player, scope:recipient = author.
# $TYPE$ must be passed as a flag, e.g. TYPE = flag:vassal_memorial.
# On success, sets the current slot and its checkpoint-based delivery ID on scope:actor.
# Does NOT write context, call log_gamedata, bump checkpoint, schedule events, or set cooldowns.
votc_allocate_incoming_letter_slot_effect = {
	# For deathbed_letter with sender_role=deceased, the letter's perceived
	# sender (shown as portrait/title in-game) should be the deceased, not the
	# mourner who only carried the letter. Pick the right target once here.
	if = {
		limit = {
			scope:actor = {
				has_variable = votc_sender_role
				var:votc_sender_role = flag:deceased
			}
		}
		set_global_variable = {
			name = votc_incoming_letter_snapshot_sender
			value = global_var:votc_incoming_letter_fact_deceased
		}
	}
	else = {
		set_global_variable = {
			name = votc_incoming_letter_snapshot_sender
			value = scope:recipient
		}
	}

	if = {
		limit = { NOT = { exists = global_var:votc_incoming_letter_1 } }
		set_global_variable = {
			name = votc_incoming_letter_1
			value = flag:occupied
		}
		set_global_variable = {
			name = votc_incoming_letter_sender_1
			value = global_var:votc_incoming_letter_snapshot_sender
		}
		set_global_variable = {
			name = votc_incoming_letter_type_1
			value = $TYPE$
		}
		scope:actor = {
			set_variable = {
				name = votc_incoming_letter_1_timer
				days = votc_letter_delay
			}
			set_variable = {
				name = votc_incoming_letter_1_sent_day
				value = current_date
			}

				name = votc_incoming_letter_1_delivery_id

			}
		}

		set_global_variable = {
			name = votc_incoming_letter_2

		}

			name = votc_incoming_letter_sender_2

		}

		}

			set_variable = {
				name = votc_incoming_letter_2_timer

			}
			set_variable = {
				name = votc_incoming_letter_2_sent_day
				value = current_date
			}
			set_variable = {
				name = votc_incoming_letter_2_delivery_id
				value = var:votc_checkpoint_epoch
			}
			set_variable = {
				name = votc_incoming_letter_current_slot
				value = 2
			}
		}
	}
	else_if = {
		limit = { NOT = { exists = global_var:votc_incoming_letter_3 } }
		set_global_variable = {
			name = votc_incoming_letter_3
			value = flag:occupied
		}
		set_global_variable = {
			name = votc_incoming_letter_sender_3
			value = global_var:votc_incoming_letter_snapshot_sender
		}
		set_global_variable = {
			name = votc_incoming_letter_type_3
			value = $TYPE$
		}
		scope:actor = {
			set_variable = {
				name = votc_incoming_letter_3_timer
				days = votc_letter_delay
			}
			set_variable = {
				name = votc_incoming_letter_3_sent_day
				value = current_date
			}
			set_variable = {
				name = votc_incoming_letter_3_delivery_id
				value = var:votc_checkpoint_epoch
			}
			set_variable = {
				name = votc_incoming_letter_current_slot
				value = 3
			}
		}
	}
	else_if = {
		limit = { NOT = { exists = global_var:votc_incoming_letter_4 } }
		set_global_variable = {
			name = votc_incoming_letter_4
			value = flag:occupied
		}
		set_global_variable = {
			name = votc_incoming_letter_sender_4
			value = global_var:votc_incoming_letter_snapshot_sender
		}
		set_global_variable = {
			name = votc_incoming_letter_type_4
			value = $TYPE$
		}
		scope:actor = {
			set_variable = {
				name = votc_incoming_letter_4_timer
				days = votc_letter_delay
			}
			set_variable = {
				name = votc_incoming_letter_4_sent_day
				value = current_date
			}
			set_variable = {
				name = votc_incoming_letter_4_delivery_id
				value = var:votc_checkpoint_epoch
			}
			set_variable = {
				name = votc_incoming_letter_current_slot

		}
	}
	else_if = {
		limit = { NOT = { exists = global_var:votc_incoming_letter_5 } }
		set_global_variable = {
			name = votc_incoming_letter_5
			value = flag:occupied

									}
								}
								else_if = {
									limit = { $TYPE$ = flag:great_project_report }
									set_variable = {
										name = votc_incoming_letter_great_project_cooldown
										days = 90
									}
								}
							}
							scope:recipient = {
								set_variable = {
									name = votc_incoming_letter_sent_recently
									days = 365
								}
							}

						# Clean up the transient slot marker.
						scope:actor = {
							remove_variable ?= votc_incoming_letter_current_slot
							remove_variable ?= votc_sender_role
							remove_variable ?= votc_death_category
						}
						}
						else = {
							# Allocation failed; the allocation effect has already logged skipped/full.
						}
					}
					else = {
						debug_log = "VOTC:INCOMING_LETTER/;/skipped/;/assert/theme_cooldown"
					}
				}
				else = {
					debug_log = "VOTC:INCOMING_LETTER/;/skipped/;/assert/global_cooldown"
				}
			}
			else = {
				debug_log = "VOTC:INCOMING_LETTER/;/skipped/;/assert/author_invalid"
			}
		}
		else = {
			debug_log = "VOTC:INCOMING_LETTER/;/skipped/;/assert/not_enabled"
		}
	}
	else = {
		debug_log = "VOTC:INCOMING_LETTER/;/skipped/;/assert/actor_is_ai"
	}
}
