# Append one random bit to the global campaign id segment variable. The

votc_append_campaign_id_bit_effect = {

# Generate all four campaign id segments into their committed globals. Only
# call after votc_validate_campaign_id_effect has cleared any half-committed
# state. Writes the schema commit flag last so a crash between segments is
# detected as corrupt rather than silently re-keyed.
votc_generate_campaign_id_effect = {
	votc_generate_campaign_id_segment_effect = yes
	set_global_variable = {
		name = votc_campaign_id_a
		value = { value = global_var:votc_campaign_id_segment }
	}
	votc_generate_campaign_id_segment_effect = yes
	set_global_variable = {
		name = votc_campaign_id_b
		value = { value = global_var:votc_campaign_id_segment }
	}
	votc_generate_campaign_id_segment_effect = yes
	set_global_variable = {
		name = votc_campaign_id_c
		value = { value = global_var:votc_campaign_id_segment }
	}
	votc_generate_campaign_id_segment_effect = yes
	set_global_variable = {
		name = votc_campaign_id_d
		value = { value = global_var:votc_campaign_id_segment }
	}
	remove_global_variable = votc_campaign_id_segment
}

# Detect corruption in an already-committed campaign id: schema is set but any
# segment is missing or zero. Must not silently regenerate; returns via the
# votc_campaign_corrupt global flag for the init effect to log.
votc_validate_campaign_id_effect = {
	set_global_variable = { name = votc_campaign_corrupt value = 0 }
	if = {
		limit = { NOT = { has_global_variable = votc_campaign_id_a } }
		set_global_variable = { name = votc_campaign_corrupt value = 1 }
	}
	if = {
		limit = { NOT = { has_global_variable = votc_campaign_id_b } }
		set_global_variable = { name = votc_campaign_corrupt value = 1 }
	}
	if = {
		limit = { NOT = { has_global_variable = votc_campaign_id_c } }
		set_global_variable = { name = votc_campaign_corrupt value = 1 }
	}
	if = {
		limit = { NOT = { has_global_variable = votc_campaign_id_d } }
		set_global_variable = { name = votc_campaign_corrupt value = 1 }
	}
	if = {
		limit = { global_var:votc_campaign_id_a <= 0 }
		set_global_variable = { name = votc_campaign_corrupt value = 1 }
	}
	if = {
		limit = { global_var:votc_campaign_id_b <= 0 }
		set_global_variable = { name = votc_campaign_corrupt value = 1 }
	}
	if = {
		limit = { global_var:votc_campaign_id_c <= 0 }
		set_global_variable = { name = votc_campaign_corrupt value = 1 }
	}
	if = {
		limit = { global_var:votc_campaign_id_d <= 0 }
		set_global_variable = { name = votc_campaign_corrupt value = 1 }
	}
}

# Idempotent campaign identity bootstrap. Order:
#  1. If schema and all four segments are intact, return.
#  2. If schema exists but any segment is missing/zero, log corrupt and return
#     (never silently overwrite a committed identity).
#  3. If schema is absent, clear any half-committed segments, set bootstrap kind
#     from whether the current player has old checkpoint evidence, generate all
#     four segments, write schema=1 as the commit marker, and log created.
# Global variables persist with the save, so a saved campaign keeps its id
# across reloads and save branches.
votc_init_campaign_id_effect = {
	# Clean any leftover scratch state from a crashed generation so the next
	# bootstrap starts fresh. ?= only removes if present, so this is a no-op on
	# a clean save.
	remove_global_variable ?= votc_campaign_id_segment
	if = {
		limit = {
			has_global_variable = votc_campaign_schema
			global_var:votc_campaign_schema >= 1
			has_global_variable = votc_campaign_id_a
			has_global_variable = votc_campaign_id_b
			has_global_variable = votc_campaign_id_c
			has_global_variable = votc_campaign_id_d
			global_var:votc_campaign_id_a > 0
			global_var:votc_campaign_id_b > 0
			global_var:votc_campaign_id_c > 0
			global_var:votc_campaign_id_d > 0
		}
		# Already committed and intact. Clear any stale corrupt flag left by a
		# previously-detected-then-manually-fixed corruption so it does not
		# linger as global state on a healthy save.
		remove_global_variable ?= votc_campaign_corrupt
	}
	else_if = {
		limit = {
			has_global_variable = votc_campaign_schema
			global_var:votc_campaign_schema >= 1
		}
		votc_validate_campaign_id_effect = yes
		debug_log = "VOTC:CAMPAIGN/;/corrupt/;/[GetPlayer.GetID]/;/[GetGlobalVariable('votc_campaign_schema').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_a').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_b').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_c').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_d').GetValue|0]/;/[GetGlobalVariable('votc_campaign_corrupt').GetValue|0]"
	}
	else = {
		set_global_variable = { name = votc_campaign_bootstrap_kind value = 1 }
		if = {
			limit = {
				OR = {
					has_variable = votc_checkpoint_epoch
					has_variable = votc_checkpoint_token
					has_variable = votc_timeline_node_a
				}
			}
			set_global_variable = { name = votc_campaign_bootstrap_kind value = 2 }
		}
		remove_global_variable = votc_campaign_id_a
		remove_global_variable = votc_campaign_id_b
		remove_global_variable = votc_campaign_id_c
		remove_global_variable = votc_campaign_id_d
		votc_generate_campaign_id_effect = yes
		set_global_variable = { name = votc_campaign_schema value = 1 }
		debug_log = "VOTC:CAMPAIGN/;/created/;/[GetPlayer.GetID]/;/[GetGlobalVariable('votc_campaign_schema').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_a').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_b').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_c').GetValue|0]/;/[GetGlobalVariable('votc_campaign_id_d').GetValue|0]/;/[GetGlobalVariable('votc_campaign_bootstrap_kind').GetValue|0]"
	}
}

votc_init_checkpoint_effect = {
	votc_init_campaign_id_effect = yes
	if = {
		limit = { NOT = { has_variable = votc_checkpoint_epoch } }
		set_variable = { name = votc_checkpoint_epoch value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = votc_timeline_node_a } }
		set_variable = { name = votc_timeline_node_a value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = votc_timeline_node_b } }
		set_variable = { name = votc_timeline_node_b value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = votc_timeline_parent_a } }
		set_variable = { name = votc_timeline_parent_a value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = votc_timeline_parent_b } }
		set_variable = { name = votc_timeline_parent_b value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = votc_checkpoint_token } }
		set_variable = { name = votc_checkpoint_token value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = votc_checkpoint_pending_token } }
		votc_generate_pending_checkpoint_token_effect = yes
	}
}

# 推进 VOTC checkpoint epoch 并记录日期。
# 用于信件/战报等流程在写入 init 日志前把 epoch 推进到下一个值，
# 避免 app 侧用旧 epoch 判定时把刚写入的 epoch+1 总结误判为"未来"而归档。
votc_bump_checkpoint_effect = {
	votc_init_checkpoint_effect = yes
	set_variable = {
		name = votc_checkpoint_token
		value = var:votc_checkpoint_pending_token
	}
	remove_variable ?= votc_checkpoint_pending_token
	change_variable = {
		name = votc_checkpoint_epoch
		add = 1
	}
	set_variable = {
		name = votc_checkpoint_day
		value = current_date
	}
	debug_log = "VOTC:CHECKPOINT/;/bump/;/[THIS.Char.GetID]/;/[THIS.Var('votc_checkpoint_epoch').GetValue|0]/;/[GetCurrentDate.GetStringShort]"
	debug_log = "VOTC:TIMELINE/;/bump/;/2/;/1/;/[THIS.Char.GetID]/;/$SOURCE$/;/[THIS.Var('votc_checkpoint_epoch').GetValue|0]/;/[THIS.Var('votc_timeline_node_a').GetValue|0]/;/[THIS.Var('votc_timeline_node_b').GetValue|0]/;/[THIS.Var('votc_timeline_parent_a').GetValue|0]/;/[THIS.Var('votc_timeline_parent_b').GetValue|0]/;/[THIS.Var('votc_checkpoint_token').GetValue|0]/;/[GetCurrentDate.GetStringShort]"
}
