# VOTC incoming letter test: liege/overlord letter.
# Console command: run votc_test_incoming_liege_letter.txt
# Copy to: C:\Users\A\Documents\Paradox Interactive\Crusader Kings III\run
# Requires the player to have a liege.

save_scope_as = actor

if = {
	limit = { exists = liege }
	liege = {
		limit = {
			votc_incoming_letter_author_valid_trigger = yes
		}
		save_scope_as = recipient
	}

	if = {
		limit = { exists = scope:recipient }
		votc_start_incoming_letter_effect = { TYPE = flag:liege_overlord_letter }
	}
	else = {
		debug_log = "VOTC:INCOMING_LETTER/;/test/;/skipped/;/invalid_liege"
	}
}
else = {
	debug_log = "VOTC:INCOMING_LETTER/;/test/;/skipped/;/no_liege"
}
