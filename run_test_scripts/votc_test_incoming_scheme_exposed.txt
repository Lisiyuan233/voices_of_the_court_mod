# Selects a random courtier as witness and fakes scheme facts.

save_scope_as = actor

random_courtier = {
	limit = {
		votc_incoming_letter_scheme_witness_valid_trigger = yes
		NOT = { this = root }
	}
	save_scope_as = recipient
}

if = {
	limit = { exists = scope:recipient }
	set_global_variable = {
		name = votc_incoming_letter_fact_scheme_owner
		value = scope:recipient
	}
	set_variable = { name = votc_scheme_type value = flag:murder }
	votc_start_incoming_letter_effect = { TYPE = flag:scheme_exposed }
}
else = {
	debug_log = "VOTC:INCOMING_LETTER/;/test/;/skipped/;/no_witness"
}
