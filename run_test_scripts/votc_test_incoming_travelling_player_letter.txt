	limit = { is_travelling = yes }
	scope:actor = {
		votc_select_family_letter_author_effect = yes

	if = {
		limit = { exists = scope:recipient }
		votc_start_family_incoming_letter_effect = yes
	}
	else = {
		debug_log = "VOTC:INCOMING_LETTER/;/test/;/skipped/;/no_valid_family"
	}
}

	debug_log = "VOTC:INCOMING_LETTER/;/test/;/skipped/;/player_not_travelling"
