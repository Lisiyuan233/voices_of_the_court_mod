								remove_variable ?= votc_battle_report_delay_days
								remove_variable ?= votc_battle_report_percent_enemies_killed
							}
						}
						else = {
							debug_log = "VOTC:BATTLE_REPORT/;/skipped/;/full/;/[GetCurrentDate.GetStringShort]"
							scope:votc_battle_report_player = {
								remove_variable ?= votc_battle_report_delay_days
								remove_variable ?= votc_battle_report_percent_enemies_killed
							}
						}
					}
					else = {
						debug_log = "VOTC:BATTLE_REPORT/;/skipped/;/player_scope/;/[GetCurrentDate.GetStringShort]"
					}
				}
				else = {
					debug_log = "VOTC:BATTLE_REPORT/;/skipped/;/not_important/;/[GetCurrentDate.GetStringShort]"
				}
			}
			else = {
				debug_log = "VOTC:BATTLE_REPORT/;/skipped/;/no_player/;/[GetCurrentDate.GetStringShort]"
			}
		}
		else = {
			debug_log = "VOTC:BATTLE_REPORT/;/skipped/;/scopes/;/[GetCurrentDate.GetStringShort]"
		}
	}
}
